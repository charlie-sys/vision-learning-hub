PyTorch 视觉基础小实验

本实验用程序生成“亮方块在左/右”的图像做二分类，并演示注意力、VAE 重参数化和扩散加噪。无需外部数据集或预训练权重。它用于学习代码流程，不能代表真实视觉任务性能。

1. 运行方法

安装适合本机的 PyTorch，然后在项目根目录运行：

python examples/vision_basics_lab.py

也可以下载完整 Python 文件。脚本自动选择可用 CUDA 或 CPU；输出各项检查结果、训练损失和验证准确率。

2. 结构导读

函数或类 学习内容 检查点
make_data 合成图像、标签与独立划分 图像 [N,1,28,28]，标签 [N]
TinyCNN 卷积、池化和全连接 两次池化后为 7×7
train_demo DataLoader、训练、验证、检查点 清梯度→前向→反向→更新，评估时关闭梯度
concepts_demo 注意力、重参数化、扩散 权重归一化、梯度存在、张量形状正确

3. 完整代码

  1"""Self-contained PyTorch lab for the companion Chinese vision tutorial.
  2
  3Run: python vision_basics_lab.py
  4No external dataset or pretrained weights are required.
  5"""
  6
  7import tempfile
  8from pathlib import Path
  9
 10import torch
 11from torch import nn
 12from torch.utils.data import DataLoader, TensorDataset
 13
 14
 15def make_data(n: int, seed: int) -> TensorDataset:
 16    """Classify whether a bright square is on the left or right."""
 17    g = torch.Generator().manual_seed(seed)
 18    x = 0.15 * torch.randn(n, 1, 28, 28, generator=g)
 19    y = torch.randint(0, 2, (n,), generator=g)
 20    row = torch.randint(5, 18, (n,), generator=g)
 21    for i in range(n):
 22        col = 4 if int(y[i]) == 0 else 18
 23        x[i, 0, int(row[i]) : int(row[i]) + 6, col : col + 6] += 1.0
 24    return TensorDataset(x.clamp(0, 1), y)
 25
 26
 27class TinyCNN(nn.Module):
 28    def __init__(self):
 29        super().__init__()
 30        self.net = nn.Sequential(
 31            nn.Conv2d(1, 16, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
 32            nn.Conv2d(16, 32, 3, padding=1), nn.ReLU(), nn.MaxPool2d(2),
 33            nn.Flatten(), nn.Linear(32 * 7 * 7, 64), nn.ReLU(), nn.Linear(64, 2),
 34        )
 35
 36    def forward(self, x):
 37        return self.net(x)
 38
 39
 40def train_demo():
 41    torch.manual_seed(7)
 42    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
 43    train = DataLoader(make_data(800, 11), batch_size=64, shuffle=True)
 44    val = DataLoader(make_data(200, 22), batch_size=64)
 45    model = TinyCNN().to(device)
 46    opt = torch.optim.AdamW(model.parameters(), lr=1e-3)
 47    loss_fn = nn.CrossEntropyLoss()
 48    for epoch in range(3):
 49        model.train()
 50        loss_sum = 0.0
 51        for x, y in train:
 52            x, y = x.to(device), y.to(device)
 53            opt.zero_grad(set_to_none=True)
 54            logits = model(x)
 55            loss = loss_fn(logits, y)
 56            loss.backward()
 57            opt.step()
 58            loss_sum += loss.item() * len(y)
 59        model.eval()
 60        correct = total = 0
 61        with torch.inference_mode():
 62            for x, y in val:
 63                pred = model(x.to(device)).argmax(dim=1).cpu()
 64                correct += int((pred == y).sum())
 65                total += len(y)
 66        print(f"epoch={epoch + 1} train_loss={loss_sum / 800:.4f} val_acc={correct / total:.3f}")
 67    with tempfile.TemporaryDirectory() as temp_dir:
 68        out = Path(temp_dir) / "tinycnn_state.pt"
 69        torch.save(model.state_dict(), out)
 70        restored = TinyCNN().to(device)
 71        restored.load_state_dict(torch.load(out, map_location=device, weights_only=True))
 72        restored.eval()
 73        with torch.inference_mode():
 74            probe = next(iter(val))[0][:2].to(device)
 75            assert torch.allclose(model(probe), restored(probe))
 76    print(f"checkpoint round-trip passed; device={device}")
 77
 78
 79def concepts_demo():
 80    # Q, K, V: [batch, tokens, channels]. Attention output has the same shape.
 81    tokens = torch.randn(2, 4, 8)
 82    q, k, v = tokens, tokens, tokens
 83    weights = torch.softmax((q @ k.transpose(-2, -1)) / (8 ** 0.5), dim=-1)
 84    attended = weights @ v
 85    assert weights.shape == (2, 4, 4) and attended.shape == tokens.shape
 86    assert torch.allclose(weights.sum(-1), torch.ones(2, 4))
 87
 88    # VAE: sample a latent variable while retaining gradients to mu and logvar.
 89    mu = torch.zeros(3, 5, requires_grad=True)
 90    logvar = torch.zeros(3, 5, requires_grad=True)
 91    z = mu + torch.exp(0.5 * logvar) * torch.randn_like(mu)
 92    kl = -0.5 * (1 + logvar - mu.square() - logvar.exp()).sum(dim=1).mean()
 93    (z.square().mean() + kl).backward()
 94    assert mu.grad is not None and logvar.grad is not None
 95
 96    # DDPM forward process at an arbitrary timestep; not a full diffusion trainer.
 97    x0 = torch.randn(4, 1, 28, 28)
 98    alpha_bar_t = torch.tensor([0.99, 0.8, 0.5, 0.1]).view(4, 1, 1, 1)
 99    eps = torch.randn_like(x0)
100    xt = alpha_bar_t.sqrt() * x0 + (1 - alpha_bar_t).sqrt() * eps
101    assert xt.shape == x0.shape
102    print("attention, VAE reparameterization, DDPM noising: shape/gradient checks passed")
103
104
105if __name__ == "__main__":
106    concepts_demo()
107    train_demo()

4. 练习与验收

  1. 分别使用 1e-4、1e-3、1e-1 学习率，解释损失曲线差别。
  2. 删除一个池化层，先算出全连接输入维度再修改代码。
  3. 将训练样本减至 50，观察训练和验证差距。
  4. 将 alpha_bar_t 改成接近 0 或 1 的值，解释噪声占比。
  5. 解释为什么保存/恢复后同一输入应产生一致输出。

完成后回到前置知识教程，继续阅读 ViT、VAE、DDPM 和 DINO。
